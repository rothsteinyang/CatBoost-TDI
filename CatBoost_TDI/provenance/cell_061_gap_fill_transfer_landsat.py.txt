import os
import rasterio
import numpy as np
import geopandas as gpd
from rasterio.mask import mask
from catboost import CatBoostRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, r2_score

# 参数设置
LST_INVALID_THRESHOLD = -200
input_dir = r"E:\ENVI\2024_12_Cloudless\valid\Landsat\multiband\30M"
output_dir = r"E:\ENVI\2024_12_Cloudless\output\2025.4\Landsat_LST_fill_0624"
shapefile_path = r"E:\ENVI\2024_11_LST_Research\data\城市矢量数据\研究区域边界.shp"

os.makedirs(output_dir, exist_ok=True)
shp = gpd.read_file(shapefile_path)
shapes = [geometry.__geo_interface__ for geometry in shp.geometry]

# 批量处理 .dat 文件
for filename in os.listdir(input_dir):
    if filename.endswith(".dat"):
        file_path = os.path.join(input_dir, filename)
        print(f"\n处理文件: {filename}")

        with rasterio.open(file_path) as src:
            bands = src.read()
            meta = src.meta

        # 提取 LST 和指定的 9 个特征变量
        LST = bands[0]
        selected_bands = np.dstack([
            bands[1],   # NDVI
            bands[3],   # NDBI
            bands[4],   # Albedo
            bands[7],   # EULUC
            bands[8],   # DEM
            bands[13],  # AH
            bands[14],  # VNL
            bands[15],  # LCZ
            bands[16]   # TDI
        ])
        rows, cols = LST.shape
        data = np.dstack([selected_bands, LST]).reshape(-1, 10)

        # 有效样本筛选
        valid_mask = (~np.isnan(data).any(axis=1)) & (data > LST_INVALID_THRESHOLD).all(axis=1)
        valid_data = data[valid_mask]
        if len(valid_data) < 100:
            print("有效样本过少，跳过该文件。")
            continue

        features = valid_data[:, :-1]
        response = valid_data[:, -1]
        total_samples = rows * cols
        valid_samples = valid_data.shape[0]
        cloud_ratio = 1 - valid_samples / total_samples
        print(f"总样本数：{total_samples}, 有效样本数：{valid_samples}, 云覆盖率：{cloud_ratio*100:.2f}%")

        # 训练模型
        X_train, X_test, y_train, y_test = train_test_split(features, response, test_size=0.2, random_state=42)
        model = CatBoostRegressor(
            iterations=8000,
            learning_rate=0.08,
            depth=10,
            eval_metric='RMSE',
            random_seed=0,
            verbose=0,
            early_stopping_rounds=100
        )
        model.fit(X_train, y_train, eval_set=(X_test, y_test), use_best_model=True)
        y_pred_test = model.predict(X_test)
        rmse = np.sqrt(mean_squared_error(y_test, y_pred_test))
        r2 = r2_score(y_test, y_pred_test)
        print(f"模型验证: RMSE={rmse:.2f}, R²={r2:.2f}")

        # 读取并裁剪 shapefile 范围
        with rasterio.open(file_path) as src:
            cropped_bands, cropped_transform = mask(src, shapes, crop=True, nodata=LST_INVALID_THRESHOLD)

        cropped_LST = cropped_bands[0]
        cropped_features = np.dstack([
            cropped_bands[1],
            cropped_bands[3],
            cropped_bands[4],
            cropped_bands[7],
            cropped_bands[8],
            cropped_bands[13],
            cropped_bands[14],
            cropped_bands[15],
            cropped_bands[16],
        ]).reshape(-1, 9)

        # 找到无效值并进行填补
        cropped_invalid_mask = (cropped_LST.flatten() <= LST_INVALID_THRESHOLD) | np.isnan(cropped_LST.flatten())
        filled_LST = cropped_LST.flatten().copy()
        X_missing = cropped_features[cropped_invalid_mask]

        if len(X_missing) > 0:
            filled_LST[cropped_invalid_mask] = model.predict(X_missing)

        # 保存重建后的 LST 影像
        filled_LST_band = filled_LST.reshape(cropped_LST.shape)
        out_meta = src.meta.copy()
        out_meta.update({
            "driver": "GTiff",
            "count": 1,
            "height": filled_LST_band.shape[0],
            "width": filled_LST_band.shape[1],
            "transform": cropped_transform
        })

        output_filename = "fil_" + os.path.splitext(filename)[0] + ".tif"
        output_path = os.path.join(output_dir, output_filename)

        with rasterio.open(output_path, "w", **out_meta) as dst:
            dst.write(filled_LST_band, 1)

        print(f"LST 重建完成，结果保存至: {output_path}")
