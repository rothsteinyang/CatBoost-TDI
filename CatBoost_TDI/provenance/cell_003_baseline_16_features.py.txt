# 导入必要库
import os
import numpy as np
import rasterio
import matplotlib.pyplot as plt
from scipy.stats import gaussian_kde, linregress
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score
from catboost import CatBoostRegressor
import pandas as pd

# ---------------- 参数设置 ----------------
LST_INVALID_THRESHOLD = -200  # 无效值阈值
input_dir = r"E:\ENVI\2024_12_Cloudless\input\multiband_2025.5.13\multi_data"
output_dir = r"E:\ENVI\2024_12_Cloudless\output\2025.4"
scatter_plot_dir = os.path.join(output_dir, "Scatter_Plots_独立TDI")
os.makedirs(scatter_plot_dir, exist_ok=True)

# 保存误差指标的列表
metrics_list = []

# ---------------- 遍历处理每一个TIF ----------------
tif_files = [f for f in os.listdir(input_dir) if f.lower().endswith('.dat')]

for tif_file in tif_files:
    file_path = os.path.join(input_dir, tif_file)
    print(f"处理文件: {file_path}")

    # 读取多波段TIF文件
    with rasterio.open(file_path) as src:
        bands = src.read()
        meta = src.meta

    # 提取各波段
    try:
        (LST, NDVI, NDWI, NDBI, Albedo, Emissivity, 
         LC, EULUC, DEM, TH, BD, BH, SVF, AH, VNL, LCZ, TDI) = bands
    except ValueError:
        print(f"文件 {tif_file} 波段数不正确，跳过。")
        continue

    # 数据堆叠
    data = np.dstack([NDVI, NDWI, NDBI, Albedo, Emissivity, LC, 
                      EULUC, DEM, TH, BD, BH, SVF, AH, VNL, LCZ, TDI, LST]).reshape(-1, 17)

    # 筛选有效数据
    valid_mask = (~np.isnan(data).any(axis=1)) & (data > LST_INVALID_THRESHOLD).all(axis=1)
    valid_data = data[valid_mask]

    if valid_data.shape[0] < 1000:
        print(f"文件 {tif_file} 有效样本太少（{valid_data.shape[0]}个），跳过。")
        continue

    features = valid_data[:, :-1]
    response = valid_data[:, -1]

    # 云覆盖比例计算
    total_samples = 323014
    valid_samples = np.sum(valid_data)
    valid_sample_ratio = 1 - (valid_samples / total_samples)

    # 模型训练
    X_train, X_test, y_train, y_test = train_test_split(features, response, test_size=0.2, random_state=42)

    catboost_model = CatBoostRegressor(
        iterations=8000,
        learning_rate=0.08,
        depth=10,
        eval_metric='RMSE',
        random_seed=0,
        verbose=0,
        early_stopping_rounds=100
    )
    catboost_model.fit(X_train, y_train, eval_set=(X_test, y_test), use_best_model=True, plot=False)

    # 模型预测与评价
    y_pred_test = catboost_model.predict(X_test)
    global_r2 = r2_score(y_test, y_pred_test)
    global_mae = mean_absolute_error(y_test, y_pred_test)
    global_rmse = np.sqrt(mean_squared_error(y_test, y_pred_test))

    # 保存评价指标
    metrics_list.append({
        "File": tif_file,
        "R2": global_r2,
        "MAE": global_mae,
        "RMSE": global_rmse,
        "Valid Samples": valid_data.shape[0],
        "Cloud Cover Ratio": valid_sample_ratio
    })

    # 绘制并保存散点图
    actual, predicted = y_test, y_pred_test
    values = np.vstack([actual, predicted])
    kde = gaussian_kde(values, bw_method=0.8)(values)

    overall_min = min(actual.min(), predicted.min())
    overall_max = max(actual.max(), predicted.max())
    margin = (overall_max - overall_min) * 0.05
    axis_range = [overall_min - margin, overall_max + margin]

    plt.figure(figsize=(10, 8), dpi=150)
    scatter = plt.scatter(actual, predicted, c=kde, cmap='jet', s=1, alpha=0.8)
    plt.plot(axis_range, axis_range, color='red', linestyle='--',
             label=f'R²: {global_r2:.2f}\nMAE: {global_mae:.2f} K\nRMSE: {global_rmse:.2f} K', alpha=0.7)

    slope, intercept, *_ = linregress(actual, predicted)
    trend_x = np.linspace(overall_min, overall_max, 100)
    trend_y = slope * trend_x + intercept
    plt.plot(trend_x, trend_y, color='darkblue', linestyle='-', linewidth=2, label='Line of Best Fit', alpha=0.5)

    plt.xlim(axis_range)
    plt.ylim(axis_range)
    plt.gca().set_aspect('equal', adjustable='box')
    plt.xlabel('ECOSTRESS LST', fontsize=14)
    plt.ylabel('Reconstruction LST', fontsize=14)
    plt.tick_params(axis='both', which='major', labelsize=12)
    plt.colorbar(scatter, label='Density')
    plt.legend(fontsize=14)
    plt.grid(True)
    plt.tight_layout()

    plot_filename = os.path.splitext(tif_file)[0] + '_scatter.png'
    plot_save_path = os.path.join(scatter_plot_dir, plot_filename)
    plt.savefig(plot_save_path)
    plt.close()

    print(f"散点图已保存: {plot_save_path}")

# ---------------- 保存指标 Excel 表格 ----------------
metrics_df = pd.DataFrame(metrics_list)
excel_path = os.path.join(scatter_plot_dir, "model_metrics.xlsx")
metrics_df.to_excel(excel_path, index=False)
print(f"\n模型指标已保存至: {excel_path}")
print("全部文件处理完成。")
