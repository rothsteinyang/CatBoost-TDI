# 单张关键变量筛选

import os
import rasterio
import numpy as np
import shap
import pandas as pd
from catboost import CatBoostRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, r2_score

# 设置无效值阈值
LST_INVALID_THRESHOLD = -200

# 输入输出路径
file_path = r"E:\ENVI\2024_12_Cloudless\input\multiband_2025.5.13\multi_data\2019_08_09_T11_14_38.dat"
output_dir = r"E:\ENVI\2024_12_Cloudless\output\2025.4"
os.makedirs(output_dir, exist_ok=True)

# 读取多波段 TIF 文件
with rasterio.open(file_path) as src:
    bands = src.read()
    meta = src.meta

# 提取波段
LST = bands[0]
NDVI = bands[1]
NDWI = bands[2]
NDBI = bands[3]
Albedo = bands[4]
Emissivity = bands[5]
LC = bands[6]
EULUC = bands[7]
DEM = bands[8]
TH = bands[9]
BD = bands[10]
BH = bands[11]
SVF = bands[12]
AH = bands[13]
VNL = bands[14]
LCZ = bands[15]
TDI = bands[16]

# 特征名
feature_names = ['NDVI', 'NDWI', 'NDBI', 'Albedo', 'Emissivity', 'LC', 'EULUC', 'DEM',
                 'TH', 'BD', 'BH', 'SVF', 'AH', 'VNL', 'LCZ', 'TDI']

# 组装数据
rows, cols = LST.shape
data = np.dstack([NDVI, NDWI, NDBI, Albedo, Emissivity, LC, EULUC, DEM,
                  TH, BD, BH, SVF, AH, VNL, LCZ, TDI, LST]).reshape(-1, 17)

valid_mask = (~np.isnan(data).any(axis=1)) & (data > LST_INVALID_THRESHOLD).all(axis=1)
valid_data = data[valid_mask]
features = valid_data[:, :-1]
response = valid_data[:, -1]

# 划分数据集
X_train, X_test, y_train, y_test = train_test_split(
    features, response, test_size=0.2, random_state=42)

# 添加列名用于 SHAP
X_train_df = pd.DataFrame(X_train, columns=feature_names)
X_test_df = pd.DataFrame(X_test, columns=feature_names)

# 训练原始模型
model = CatBoostRegressor(
    iterations=8000,
    learning_rate=0.08,
    depth=10,
    eval_metric='RMSE',
    random_seed=0,
    verbose=200,
    early_stopping_rounds=100
)
model.fit(X_train_df, y_train, eval_set=(X_test_df, y_test), use_best_model=True, plot=False)

# 基准模型性能
y_pred = model.predict(X_test_df)
rmse = np.sqrt(mean_squared_error(y_test, y_pred))
r2 = r2_score(y_test, y_pred)
print(f"[原始模型] RMSE = {rmse:.2f} K, R² = {r2:.2f}")

# 计算 SHAP 值并获取平均重要性
explainer = shap.Explainer(model)
shap_values = explainer(X_train_df)
mean_abs_shap = np.abs(shap_values.values).mean(axis=0)
shap_ranking = pd.Series(mean_abs_shap, index=feature_names).sort_values(ascending=False)
print("\nSHAP 特征重要性排序：")
print(shap_ranking)

# 自动选择 Top-N 特征重新训练
top_n_list = [5, 8, 10, 12]  # 可以按需修改
print("\nTop-N 特征简化模型评估：")

for n in top_n_list:
    top_features = shap_ranking.head(n).index.tolist()

    X_train_sub = X_train_df[top_features]
    X_test_sub = X_test_df[top_features]

    model_sub = CatBoostRegressor(
        iterations=8000,
        learning_rate=0.08,
        depth=10,
        eval_metric='RMSE',
        random_seed=0,
        verbose=0,
        early_stopping_rounds=100
    )
    model_sub.fit(X_train_sub, y_train, eval_set=(X_test_sub, y_test), use_best_model=True, plot=False)

    y_pred_sub = model_sub.predict(X_test_sub)
    rmse_sub = np.sqrt(mean_squared_error(y_test, y_pred_sub))
    r2_sub = r2_score(y_test, y_pred_sub)

    print(f"  Top-{n} 特征模型: RMSE = {rmse_sub:.2f} K, R² = {r2_sub:.2f}")
