import rasterio
import numpy as np
from catboost import CatBoostRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, r2_score
import pandas as pd
import os
import glob
import matplotlib.pyplot as plt

# 参数
LST_INVALID_THRESHOLD = -200
input_dir = r"E:\ENVI\2024_12_Cloudless\input\multiband_2025.5.13\multi_data"
output_dir = r"E:\ENVI\2024_12_Cloudless\output\2025.4\不同缺失量模型性能对比"
os.makedirs(output_dir, exist_ok=True)

# 指定要提取的波段索引（基于0的索引）
selected_bands = [1, 3, 4, 7, 8, 13, 14, 15, 16]  # NDVI, NDBI, Albedo, EULUC, DEM, AH, VNL, LCZ, TDI

# 获取所有dat文件
dat_files = glob.glob(os.path.join(input_dir, "*.dat"))
print(f"找到 {len(dat_files)} 个dat文件")

# 处理每个dat文件
for file_path in dat_files:
    print(f"\n正在处理文件: {os.path.basename(file_path)}")
    
    try:
        # 读取DAT文件
        with rasterio.open(file_path) as src:
            bands = src.read()
        
        # 检查波段数量
        if bands.shape[0] < max(selected_bands) + 1:
            print(f"警告: 文件 {os.path.basename(file_path)} 波段数不足，跳过处理")
            continue
        
        # 分离波段
        LST = bands[0]  # LST是第一个波段
        
        # 只提取指定的特征波段
        selected_features = []
        for band_idx in selected_bands:
            selected_features.append(bands[band_idx])
        features_stack = np.dstack(selected_features)  # 只包含选定的9个波段
        
        # 构建数据矩阵
        rows, cols = LST.shape
        data = np.dstack([features_stack, LST]).reshape(-1, len(selected_bands) + 1)  # shape: (n, 10) - 9个特征 + 1个LST
        
        # 有效掩码
        valid_mask = (
            ~np.isnan(data).any(axis=1) &
            (data > LST_INVALID_THRESHOLD).all(axis=1)
        )
        valid_data = data[valid_mask]
        features_full = valid_data[:, :-1]  # 9个特征波段
        response_full = valid_data[:, -1]   # LST
        
        total_valid = len(response_full)
        print(f"有效样本总数: {total_valid}")
        
        if total_valid < 100:  # 如果有效样本太少，跳过
            print(f"有效样本数量太少 ({total_valid})，跳过此文件")
            continue
        
        # 缺失模拟比例列表
        keep_ratios = np.arange(1.0, 0.0, -0.1)
        
        results = []
        
        for ratio in keep_ratios:
            subset_size = int(total_valid * ratio)
            print(f"\n=== 当前保留比例: {int(ratio*100)}%，样本数: {subset_size} ===")
            
            if subset_size < 50:  # 如果子集太小，跳过
                print(f"子集样本数太少 ({subset_size})，跳过此比例")
                continue
            
            np.random.seed(42)
            subset_indices = np.random.choice(total_valid, subset_size, replace=False)
            
            X_subset = features_full[subset_indices]
            y_subset = response_full[subset_indices]
            
            # 构建未使用样本（模拟缺失部分）
            mask_all = np.ones(total_valid, dtype=bool)
            mask_all[subset_indices] = False
            X_missing_sim = features_full[mask_all]
            y_missing_sim = response_full[mask_all]
            
            # 划分训练集和测试集
            X_train, X_test, y_train, y_test = train_test_split(X_subset, y_subset, train_size=0.8, random_state=42)
            
            # 模型训练
            model = CatBoostRegressor(
                iterations=8000,
                learning_rate=0.08,
                depth=10,
                eval_metric='RMSE',
                random_seed=0,
                verbose=0,
                early_stopping_rounds=100
            )
            model.fit(X_train, y_train, eval_set=(X_test, y_test), use_best_model=True)
            
            # 测试集精度
            y_pred_test = model.predict(X_test)
            rmse_test = np.sqrt(mean_squared_error(y_test, y_pred_test))
            r2_test = r2_score(y_test, y_pred_test)
            
            # 模拟缺失部分预测精度
            if len(y_missing_sim) > 0:
                y_pred_missing = model.predict(X_missing_sim)
                rmse_missing = np.sqrt(mean_squared_error(y_missing_sim, y_pred_missing))
                r2_missing = r2_score(y_missing_sim, y_pred_missing)
            else:
                rmse_missing = None
                r2_missing = None
            
            print(f"[测试集] RMSE: {rmse_test:.4f}, R²: {r2_test:.4f}")
            if rmse_missing is not None:
                print(f"[模拟缺失] RMSE: {rmse_missing:.4f}, R²: {r2_missing:.4f}")
            else:
                print("[模拟缺失] 无剩余样本用于评估")
            
            # 保存结果
            results.append({
                "保留比例": f"{int(ratio*100)}%",
                "保留样本数": subset_size,
                "模拟缺失样本数": len(y_missing_sim),
                "训练样本数": len(y_train),
                "测试样本数": len(y_test),
                "测试集_RMSE(K)": round(rmse_test, 4),
                "测试集_R2": round(r2_test, 4),
                "缺失区_RMSE(K)": round(rmse_missing, 4) if rmse_missing is not None else "N/A",
                "缺失区_R2": round(r2_missing, 4) if r2_missing is not None else "N/A"
            })
        
        if not results:  # 如果没有有效结果，跳过保存
            print(f"文件 {os.path.basename(file_path)} 没有产生有效结果，跳过保存")
            continue
        
        # 从输入路径中提取基础文件名（不含扩展名）
        dat_basename = os.path.splitext(os.path.basename(file_path))[0]
        
        # 保存为Excel
        results_df = pd.DataFrame(results)
        excel_path = os.path.join(output_dir, f"{dat_basename}_不同数据缺失比例下的模型性能评估.xlsx")
        results_df.to_excel(excel_path, index=False)
        
        print(f"\n完成！结果已保存至：{excel_path}")
        
        # ---------- 绘制误差趋势图 ----------
        # 使用有效的 matplotlib 样式
        plt.style.use('ggplot')
        
        # 计算云覆盖比例（无效数据比例）
        ratios_raw = [int(r.strip('%')) for r in results_df['保留比例']]
        cloud_ratios = [100 - r for r in ratios_raw]  # 例如：100 - 90 = 10（10% 云覆盖）
        
        rmse_test = results_df['测试集_RMSE(K)']
        r2_test = results_df['测试集_R2']
        rmse_missing = results_df['缺失区_RMSE(K)'].replace("N/A", np.nan)
        r2_missing = results_df['缺失区_R2'].replace("N/A", np.nan)
        
        # 创建绘图窗口
        plt.figure(figsize=(12, 6))
        
        # --- RMSE subplot ---
        plt.subplot(1, 2, 1)
        plt.plot(cloud_ratios, rmse_test, 'o-', label='Model RMSE')
        plt.plot(cloud_ratios, rmse_missing, 's--', label='Cloud Cover Area RMSE')
        plt.xlabel("Cloud Cover Ratio (%)")
        plt.ylabel("RMSE (K)")
        plt.title("Model RMSE vs Cloud Cover RMSE")
        plt.legend()
        plt.grid(True)
        
        # --- R² subplot ---
        plt.subplot(1, 2, 2)
        plt.plot(cloud_ratios, r2_test, 'o-', label='Model R²')
        plt.plot(cloud_ratios, r2_missing, 's--', label='Cloud Cover Area R²')
        plt.xlabel("Cloud Cover Ratio (%)")
        plt.ylabel("R²")
        plt.title("Model R² vs Cloud Cover R²")
        plt.legend()
        plt.grid(True)
        
        # 构造图像保存路径
        figure_filename = f"{dat_basename}_Performance_vs_CloudCoverRatio.png"
        figure_path = os.path.join(output_dir, figure_filename)
        
        # 保存图像
        plt.tight_layout()
        plt.savefig(figure_path, dpi=300)
        plt.close()  # 关闭图形以释放内存
        
        print(f"Trend figure saved to: {figure_path}")
        
    except Exception as e:
        print(f"处理文件 {os.path.basename(file_path)} 时出错: {str(e)}")
        continue

print(f"\n所有文件处理完成！结果保存在: {output_dir}")
