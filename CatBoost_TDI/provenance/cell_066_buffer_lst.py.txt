import os
import geopandas as gpd
import rasterio
import rasterio.mask
from shapely.geometry import mapping
import numpy as np
import pandas as pd
from tqdm import tqdm
from pyproj import CRS


def get_utm_crs_from_bounds(bounds):
    """手动估算适用的UTM坐标系"""
    lon_center = (bounds.left + bounds.right) / 2
    lat_center = (bounds.top + bounds.bottom) / 2
    utm_zone = int((lon_center + 180) / 6) + 1
    is_northern = lat_center >= 0
    return CRS.from_dict({
        'proj': 'utm',
        'zone': utm_zone,
        'datum': 'WGS84',
        'south': not is_northern
    })


def reproject_raster_to_utm(src_path, dst_path):
    with rasterio.open(src_path) as src:
        utm_crs = src.crs if src.crs.is_projected else get_utm_crs_from_bounds(src.bounds)
        transform_, width, height = rasterio.warp.calculate_default_transform(
            src.crs, utm_crs, src.width, src.height, *src.bounds
        )

        kwargs = src.meta.copy()
        kwargs.update({
            'crs': utm_crs,
            'transform': transform_,
            'width': width,
            'height': height
        })

        with rasterio.open(dst_path, 'w', **kwargs) as dst:
            for i in range(1, src.count + 1):
                rasterio.warp.reproject(
                    source=rasterio.band(src, i),
                    destination=rasterio.band(dst, i),
                    src_transform=src.transform,
                    src_crs=src.crs,
                    dst_transform=transform_,
                    dst_crs=utm_crs,
                    resampling=rasterio.warp.Resampling.nearest
                )
    return dst_path, utm_crs


def reproject_vector_to_utm(shp_path, target_crs):
    gdf = gpd.read_file(shp_path)
    gdf = gdf.to_crs(target_crs)
    return gdf


def extract_mean_LST_per_buffer(gdf, raster_path, output_dir):
    with rasterio.open(raster_path) as src:
        for class_name in gdf["CLASS_NAME"].unique():
            class_df = gdf[gdf["CLASS_NAME"] == class_name]
            excel_path = os.path.join(output_dir, f"{class_name}.xlsx")
            with pd.ExcelWriter(excel_path, engine="openpyxl") as writer:
                for idx, row in tqdm(class_df.iterrows(), total=len(class_df), desc=f"Processing {class_name}"):
                    fid = row["FID"] if "FID" in row else idx
                    geometry = row.geometry
                    records = []

                    for dist in range(50, 1001, 50):
                        outer = geometry.buffer(dist)
                        inner = geometry.buffer(dist - 50)
                        ring = outer.difference(inner)

                        try:
                            out_image, _ = rasterio.mask.mask(
                                src, [mapping(ring)], crop=True, filled=False)
                            data = out_image[0]
                            valid = data[(data != src.nodata) & (~np.isnan(data))]
                            mean_val = float(np.nanmean(valid)) if valid.size > 0 else np.nan
                        except Exception:
                            mean_val = np.nan

                        records.append({"Distance": dist, "LST": mean_val})

                    df = pd.DataFrame(records)
                    df.to_excel(writer, sheet_name=f"FID_{fid}", index=False)
            print(f"Saved: {excel_path}")


def main():
    # 输入路径
    shp_path = r"E:\ENVI\2024_12_Cloudless\data\2025.4 指标重处理\TDI\扩散范围\6.8 增选\高反中心_增选1.shp"
    lst_path = r"E:\ENVI\2024_12_Cloudless\data\2025_04_arcgis_out\LST\LC-LST结果绘图用\2019_08_09_T11_14_38.tif"
    output_dir = r"E:\ENVI\2024_12_Cloudless\output\2025.4\温度扩散范围_增选"
    os.makedirs(output_dir, exist_ok=True)

    # 重投影 LST 到 UTM
    lst_utm_path = os.path.join(output_dir, "LST_UTM.tif")
    lst_utm_path, utm_crs = reproject_raster_to_utm(lst_path, lst_utm_path)

    # 重投影 shp 到 UTM
    gdf = reproject_vector_to_utm(shp_path, utm_crs)

    # 计算缓冲区温度
    extract_mean_LST_per_buffer(gdf, lst_utm_path, output_dir)


if __name__ == "__main__":
    main()
