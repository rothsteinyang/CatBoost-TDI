import rasterio
import numpy as np
import os

def write_raster(filepath, data_array, profile, nodata_val=-9999):
    """
    一个辅助函数，用于将计算出的Numpy数组写入GeoTIFF文件。
    """
    # 更新元数据以适应单波段浮点数输出
    profile.update({
        'driver': 'GTiff',
        'count': 1,
        'dtype': 'float32',
        'nodata': nodata_val
    })
    
    # 确保数据中的NaN值被替换为nodata值
    data_array[np.isnan(data_array)] = nodata_val
    
    with rasterio.open(filepath, 'w', **profile) as dst:
        dst.write(data_array.astype(np.float32), 1)

def calculate_and_save_indices(dat_file_path, output_base_dir):
    """
    主处理函数：读取dat文件，计算所有指数，并保存到指定位置。
    """
    try:
        # --- 1. 解析文件名并创建输出目录 ---
        filename = os.path.basename(dat_file_path)
        # 从 "L2A_2025_105_110151_T49QGG.dat" 中提取 "2025_105" 和 "2025_105_110151_T49QGG"
        parts = filename.replace('L2A_', '').replace('.dat', '').split('_')
        date_folder_name = f"{parts[0]}_{parts[1]}" # e.g., "2025_105"
        base_output_name = '_'.join(parts) # e.g., "2025_105_110151_T49QGG"
        
        # 创建日期专属的输出文件夹
        date_output_dir = os.path.join(output_base_dir, date_folder_name)
        os.makedirs(date_output_dir, exist_ok=True)
        
        print(f"  -> 输出将被保存至: {date_output_dir}")

        # --- 2. 读取所需的波段数据 ---
        # 我们的波段顺序是: B1, B2, B3, B4, B5, B6, B7, B8, B8A, B9, B11, B12
        # ρ2 -> band 2, ρ3 -> band 3, ρ4 -> band 4, ρ8 -> band 8, ρ11 -> band 11
        with rasterio.open(dat_file_path) as src:
            profile = src.profile # 保存元数据以供输出使用
            
            # 读取并转换为浮点数，同时进行缩放以得到真实反射率（通常除以10000）
            # L2A产品的像元值是整型，需要缩放为0-1范围的反射率才能进行科学计算
            scale_factor = 10000.0
            
            rho2 = src.read(2).astype('float32') / scale_factor
            rho3 = src.read(3).astype('float32') / scale_factor
            rho4 = src.read(4).astype('float32') / scale_factor
            rho8 = src.read(8).astype('float32') / scale_factor
            rho11 = src.read(11).astype('float32') / scale_factor
        
        # --- 3. 计算指数 ---
        # 忽略计算中可能出现的除零错误 (e.g., 0/0)
        np.seterr(divide='ignore', invalid='ignore')

        # NDVI
        print("    - 计算 NDVI...")
        ndvi = (rho8 - rho4) / (rho8 + rho4)

        # NDWI
        print("    - 计算 NDWI...")
        ndwi = (rho3 - rho8) / (rho3 + rho8)

        # NDBI
        print("    - 计算 NDBI...")
        ndbi = (rho11 - rho8) / (rho11 + rho8)
        
        # Albedo
        print("    - 计算 Albedo...")
        albedo = 0.443 * rho2 + 0.317 * rho3 + 0.240 * rho4

        # --- 4. 保存结果 ---
        # NDVI
        ndvi_path = os.path.join(date_output_dir, f"NDVI_{base_output_name}.tif")
        write_raster(ndvi_path, ndvi, profile)
        print(f"      √ NDVI 已保存: {os.path.basename(ndvi_path)}")
        
        # NDWI
        ndwi_path = os.path.join(date_output_dir, f"NDWI_{base_output_name}.tif")
        write_raster(ndwi_path, ndwi, profile)
        print(f"      √ NDWI 已保存: {os.path.basename(ndwi_path)}")

        # NDBI
        ndbi_path = os.path.join(date_output_dir, f"NDBI_{base_output_name}.tif")
        write_raster(ndbi_path, ndbi, profile)
        print(f"      √ NDBI 已保存: {os.path.basename(ndbi_path)}")

        # Albedo
        albedo_path = os.path.join(date_output_dir, f"Albedo_{base_output_name}.tif")
        write_raster(albedo_path, albedo, profile)
        print(f"      √ Albedo 已保存: {os.path.basename(albedo_path)}")

        return True

    except Exception as e:
        print(f"  -> 处理文件 {os.path.basename(dat_file_path)} 时发生错误: {e}")
        return False

# --- 主程序 ---
if __name__ == "__main__":
    # 1. 设置包含 .dat 文件的输入目录
    #    这个目录就是上一个脚本的输出目录
    stacked_dat_dir = r"E:\ENVI\2024_11_LST_Research\download\sentinel2\resampling\stacked_dat_output"

    # 2. 设置一个用于存放所有指数结果的根目录
    #    脚本会自动在此目录下创建日期子文件夹
    indices_output_dir = r"E:\ENVI\2024_11_LST_Research\download\sentinel2\resampling\indices_output"

    # 确保主输出目录存在
    os.makedirs(indices_output_dir, exist_ok=True)

    print(f"开始扫描输入目录: {stacked_dat_dir}")
    print(f"所有指数产品将保存在: {indices_output_dir}\n")

    # 查找所有 .dat 文件
    dat_files = [f for f in os.listdir(stacked_dat_dir) if f.lower().endswith('.dat')]

    if not dat_files:
        print("错误：在输入目录中没有找到 .dat 文件！请检查路径。")
    else:
        processed_count = 0
        failed_count = 0
        for dat_filename in dat_files:
            print(f"正在处理文件: {dat_filename}")
            full_path = os.path.join(stacked_dat_dir, dat_filename)
            success = calculate_and_save_indices(full_path, indices_output_dir)
            if success:
                processed_count += 1
            else:
                failed_count += 1
            print("-" * 70)
        
        print("\n🎉 全部处理完成！")
        print(f"总计：成功处理 {processed_count} 个 .dat 文件，失败 {failed_count} 个。")
