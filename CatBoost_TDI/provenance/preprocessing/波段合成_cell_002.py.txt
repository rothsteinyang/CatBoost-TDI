import os
import re
import rasterio
from rasterio.io import MemoryFile
from rasterio.warp import reproject, Resampling
from rasterio.mask import mask
import geopandas as gpd
import numpy as np

# --- 1. 配置 ---
base_dir = r"E:\ENVI\2025_08_CNN_CatBoost\input\variables_0921"
shp_path = os.path.join(base_dir, "广州边界.shp")
lst_dir = os.path.join(base_dir, "LST")
feature_dir = os.path.join(base_dir, "feature_9_16")

output_dir = os.path.join(base_dir, "ECO_for_Ta_stacked")
os.makedirs(output_dir, exist_ok=True)
print(f"输出文件将保存到: {output_dir}")

# 特征名称（与合成波段顺序一致，LST 在最前）
feature_names = ["NDVI", "NDWI", "NDBI", "Albedo", "DEM", "EULUC", "LCZ", "AH", "VNL"]

# 对应的重采样方法
resampling_methods = {
    "LST": "NEAREST",
    "NDVI": "BILINEAR",
    "NDWI": "BILINEAR",
    "NDBI": "BILINEAR",
    "Albedo": "BILINEAR",
    "DEM": "BILINEAR",
    "EULUC": "NEAREST",
    "LCZ": "NEAREST",
    "AH": "CUBIC",
    "VNL": "CUBIC"
}
resampling_map = {
    "NEAREST": Resampling.nearest,
    "BILINEAR": Resampling.bilinear,
    "CUBIC": Resampling.cubic
}

# --- 2. 读取裁剪边界 ---
try:
    shp = gpd.read_file(shp_path)
    print("裁剪边界 Shapefile 加载成功。")
except Exception as e:
    print(f"错误: 无法加载 Shapefile '{shp_path}'. 错误信息: {e}")
    exit()

# --- 3. 遍历所有 LST 文件 ---
for lst_file in os.listdir(lst_dir):
    if not lst_file.lower().startswith("lst_") or not lst_file.lower().endswith(".tif"):
        continue

    print(f"\n--- 开始处理 LST 文件: {lst_file} ---")
    lst_path = os.path.join(lst_dir, lst_file)

    match = re.match(r"LST_(\d{4})_\d{3}_\d{6}\.tif", lst_file, re.IGNORECASE)
    if not match:
        print(f"跳过未匹配命名规则的文件: {lst_file}")
        continue
    year = match.group(1)
    print(f"检测到年份: {year}")

    # --- 3.1 读取 LST 作为参考 ---
    try:
        with rasterio.open(lst_path) as ref:
            ref_profile = ref.profile
            ref_crs = ref.crs
            
            if shp.crs != ref_crs:
                print(f"警告: Shapefile 的坐标系 ({shp.crs}) 与栅格 ({ref_crs}) 不同，正在进行转换...")
                shp = shp.to_crs(ref_crs)

            ref_transform = ref.transform
            ref_width = ref.width
            ref_height = ref.height

            bands_to_stack = [ref.read(1)]
            print("LST 波段已读取作为基准。")

    except Exception as e:
        print(f"错误: 无法读取或处理基准LST文件 '{lst_path}'. 错误信息: {e}")
        continue

    # --- 3.2 匹配并重采样所有 feature 文件 ---
    feature_year_dir = os.path.join(feature_dir, year)
    if not os.path.exists(feature_year_dir):
        print(f"警告: 未找到年份 {year} 的特征文件夹，后续波段将为空: {feature_year_dir}")
        for _ in feature_names:
            bands_to_stack.append(np.full((ref_height, ref_width), np.nan, dtype=np.float32))
    else:
        for fname in feature_names:
            matches = [f for f in os.listdir(feature_year_dir) if f.startswith(fname + "_") and f.endswith(".tif")]
            
            if not matches:
                print(f"警告: 在 {year} 年未找到 {fname} 的文件，将使用空值填充。")
                bands_to_stack.append(np.full((ref_height, ref_width), np.nan, dtype=np.float32))
                continue

            feature_path = os.path.join(feature_year_dir, matches[0])
            method = resampling_methods.get(fname, "BILINEAR") # 使用 .get() 增加稳健性
            print(f"正在重采样: {fname} (方法: {method})")

            try:
                with rasterio.open(feature_path) as src:
                    resampled_data = np.empty((ref_height, ref_width), dtype=np.float32)
                    reproject(
                        source=rasterio.band(src, 1),
                        destination=resampled_data,
                        src_transform=src.transform,
                        src_crs=src.crs,
                        dst_transform=ref_transform,
                        dst_crs=ref_crs,
                        resampling=resampling_map[method]
                    )
                    bands_to_stack.append(resampled_data)
            except Exception as e:
                print(f"错误: 处理文件 '{feature_path}' 时出错，将使用空值填充。错误信息: {e}")
                bands_to_stack.append(np.full((ref_height, ref_width), np.nan, dtype=np.float32))

    # --- 4. 一次性堆叠与掩膜 ---
    if len(bands_to_stack) != len(feature_names) + 1:
        print(f"错误: 预期波段数为 {len(feature_names) + 1}, 实际收集到 {len(bands_to_stack)}。跳过此年份。")
        continue

    stacked_array = np.stack(bands_to_stack, axis=0)
    print(f"\n所有 {stacked_array.shape[0]} 个波段已重采样并堆叠。")

    temp_profile = ref_profile.copy()
    temp_profile.update({
        'count': stacked_array.shape[0],
        'dtype': stacked_array.dtype,
        'nodata': np.nan
    })

    print("正在对所有波段执行一次性掩膜操作...")
    try:
        # 使用 with 语句来正确处理内存文件
        with MemoryFile() as memfile:
            with memfile.open(**temp_profile) as dataset: # 1. 打开内存数据集并传入profile
                dataset.write(stacked_array)              # 2. 写入数据
                # 3. 将内存数据集对象传递给 mask 函数
                masked_data, masked_transform = mask(
                    dataset,
                    shp.geometry,
                    crop=True,
                    all_touched=True,
                    nodata=np.nan
                )
        print("掩膜操作完成。")
    except Exception as e:
        print(f"错误: 执行掩膜时出错。错误信息: {e}")
        continue

    # --- 5. 保存合成后的多波段 dat 文件 ---
    output_basename = os.path.splitext(lst_file)[0].replace("LST_", "") # 从LST文件名中提取基础名
    output_path = os.path.join(output_dir, f"{output_basename}.dat") # 构建输出路径
    
    output_profile = ref_profile.copy()
    output_profile.update({
        "driver": "ENVI",
        "count": masked_data.shape[0],
        "height": masked_data.shape[1],
        "width": masked_data.shape[2],
        "transform": masked_transform,
        "dtype": np.float32,
        "nodata": np.nan
    })

    with rasterio.open(output_path, "w", **output_profile) as dst:
        dst.write(masked_data)

    print(f"✅ 处理完成，输出文件: {output_path}")
