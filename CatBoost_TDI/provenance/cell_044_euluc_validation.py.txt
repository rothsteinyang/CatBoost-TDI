import os
import rasterio
import numpy as np
import pandas as pd
from catboost import CatBoostRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, r2_score

# 设置路径
input_dir = r"E:\ENVI\2024_12_Cloudless\input\multiband_2025.5.13\multi_data"
output_dir = r"E:\ENVI\2024_12_Cloudless\output\2025.4\EULUC分类精度评估"
os.makedirs(output_dir, exist_ok=True)

LST_INVALID_THRESHOLD = -200

# EULUC 编号与名称映射
euluc_class_names = {
    101: "Residential",
    201: "Business office",
    202: "Commercial service",
    301: "Industrial",
    401: "Road",
    402: "Transportation stations",
    403: "Airport facilities",
    501: "Administrative",
    502: "Educational",
    503: "Medical",
    504: "Sport & cultural",
    505: "Park & greenspace",
    1100: "Water"
}

# 存储所有结果
all_results = []

# 遍历文件夹中的所有 .dat 文件
for file_name in os.listdir(input_dir):
    if not file_name.endswith(".dat"):
        continue

    file_path = os.path.join(input_dir, file_name)
    print(f"\n正在处理文件：{file_name}")

    try:
        with rasterio.open(file_path) as src:
            bands = src.read()
            meta = src.meta

        # 提取需要的波段
        LST = bands[0]
        NDVI = bands[1]
        NDBI = bands[3]
        Albedo = bands[4]
        EULUC = bands[7]
        DEM = bands[8]
        AH = bands[13]
        VNL = bands[14]
        LCZ = bands[15]
        TDI = bands[16]

        # 堆叠数据
        rows, cols = LST.shape
        data = np.dstack([NDVI, NDBI, Albedo, EULUC, DEM, AH, VNL, LCZ, TDI, LST]).reshape(-1, 10)

        # 筛选有效值
        valid_mask = (
            ~np.isnan(data).any(axis=1) &
            (data > LST_INVALID_THRESHOLD).all(axis=1)
        )
        valid_data = data[valid_mask]

        if valid_data.shape[0] < 100:
            print(f"{file_name} 有效像元不足，跳过")
            continue

        features = valid_data[:, :-1]
        response = valid_data[:, -1]

        # 拆分训练与测试集
        X_train, X_test, y_train, y_test = train_test_split(
            features, response, test_size=0.2, random_state=42
        )

        # 训练模型
        model = CatBoostRegressor(
            iterations=8000,
            learning_rate=0.08,
            depth=10,
            eval_metric='RMSE',
            random_seed=0,
            verbose=0,
            early_stopping_rounds=100
        )
        model.fit(X_train, y_train, eval_set=(X_test, y_test), use_best_model=True)

        # 总体测试集性能
        y_pred_test = model.predict(X_test)
        rmse = np.sqrt(mean_squared_error(y_test, y_pred_test))
        r2 = r2_score(y_test, y_pred_test)
        print(f"总体 RMSE = {rmse:.2f} K, R² = {r2:.2f}")

        # -------- 关键修改：仅使用测试集数据计算类别RMSE --------
        # 从测试集特征中提取EULUC类别（索引3对应第4个特征）
        euluc_test = X_test[:, 3]
        unique_euluc_classes = np.unique(euluc_test)

        for euluc_class in unique_euluc_classes:
            # 筛选当前类别的测试样本
            class_mask = (euluc_test == euluc_class)
            y_class_true = y_test[class_mask]
            y_class_pred = y_pred_test[class_mask]

            if len(y_class_true) == 0:
                continue  # 跳过空类别

            class_rmse = np.sqrt(mean_squared_error(y_class_true, y_class_pred))
            class_r2 = r2_score(y_class_true, y_class_pred)

            euluc_name = euluc_class_names.get(int(euluc_class), "Unknown")
            all_results.append({
                "文件名": file_name,
                "EULUC类别": int(euluc_class),
                "EULUC名称": euluc_name,
                "样本数": len(y_class_true),
                "RMSE": round(class_rmse, 2),
                "R2": round(class_r2, 2)
            })
            
            print(f"EULUC 类别 {int(euluc_class)}: 样本数 = {len(y_class_true)}, RMSE = {class_rmse:.2f} K, R² = {class_r2:.2f}")

    except Exception as e:
        print(f"{file_name} 处理失败：{e}")

# 保存最终结果
results_df = pd.DataFrame(all_results)
output_csv = os.path.join(output_dir, "EULUC分类模型精度.csv")
results_df.to_csv(output_csv, index=False, encoding='utf-8-sig')

print("\n所有文件处理完成，结果保存至：", output_csv)
